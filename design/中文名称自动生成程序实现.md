### 3.2 名称生成器核心类 (generator.py)

```python
import random
import os
from typing import List, Dict
from app.config import config
from app.uniqueness import is_unique, add_to_unique
from app.quality import evaluate_name_quality

class NameGenerator:
    def __init__(self, style: str, length: int = 2, count: int = 1):
        if style not in config.STYLES:
            raise ValueError(f"Invalid style: {style}. Valid styles: {config.STYLES}")
        
        if not (config.MIN_LENGTH <= length <= config.MAX_LENGTH):
            raise ValueError(f"Length must be between {config.MIN_LENGTH} and {config.MAX_LENGTH}")
        
        self.style = style
        self.length = length
        self.count = count
        self.used_names = set()
        self.word库 = self._load_word库()
    
    def _load_word库(self) -> Dict[str, List[str]]:
        """加载词库"""
        word库 = {
            "prefix": [],
            "middle": [],
            "suffix": []
        }
        
        for part in word库.keys():
            file_path = os.path.join(config.WORD_LIB_PATH, part, f"{self.style}.txt")
            if os.path.exists(file_path):
                with open(file_path, "r", encoding="utf-8") as f:
                    word库[part] = [line.strip() for line in f if line.strip()]
        
        return word库
    
    def _generate_two_character(self) -> str:
        """生成双字名"""
        if self.style == "internet":
            # 互联网风格：姓氏 + 名字
            if self.word库["prefix"] and self.word库["suffix"]:
                return random.choice(self.word库["prefix"]) + random.choice(self.word库["suffix"])
        
        # 其他风格：前缀 + 后缀
        if self.word库["prefix"] and self.word库["suffix"]:
            return random.choice(self.word库["prefix"]) + random.choice(self.word库["suffix"])
        elif self.word库["suffix"]:
            return random.choice(self.word库["suffix"]) + random.choice(self.word库["suffix"])
        else:
            raise ValueError(f"Insufficient word库 for {self.style} style")
    
    def _generate_three_character(self) -> str:
        """生成三字名"""
        if self.style == "wuxia":
            # 武侠风格：复姓 + 单名
            if self.word库["prefix"] and self.word库["suffix"]:
                return random.choice(self.word库["prefix"]) + random.choice(self.word库["suffix"])
        
        # 其他风格：前缀 + 中间词 + 后缀
        if self.word库["prefix"] and self.word库["middle"] and self.word库["suffix"]:
            return (random.choice(self.word库["prefix"]) + 
                    random.choice(self.word库["middle"]) + 
                    random.choice(self.word库["suffix"]))
        elif self.word库["prefix"] and self.word库["suffix"]:
            return random.choice(self.word库["prefix"]) + random.choice(self.word库["suffix"]) * 2
        else:
            raise ValueError(f"Insufficient word库 for {self.style} style")
    
    def _generate_four_character(self) -> str:
        """生成四字名"""
        if self.style == "wuxia":
            # 武侠风格：复姓 + 双名
            if self.word库["prefix"] and self.word库["suffix"]:
                return random.choice(self.word库["prefix"]) + 
                       random.choice(self.word库["suffix"]) + 
                       random.choice(self.word库["suffix"])
        
        # 其他风格：前缀 + 中间词 + 后缀
        if self.word库["prefix"] and self.word库["middle"] and self.word库["suffix"]:
            return (random.choice(self.word库["prefix"]) + 
                    random.choice(self.word库["middle"]) + 
                    random.choice(self.word库["middle"]) + 
                    random.choice(self.word库["suffix"]))
        elif self.word库["prefix"] and self.word库["suffix"]:
            return random.choice(self.word库["prefix"]) + random.choice(self.word库["suffix"]) * 3
        else:
            raise ValueError(f"Insufficient word库 for {self.style} style")
    
    def _generate_single_name(self) -> str:
        """生成单个名称"""
        generators = {
            2: self._generate_two_character,
            3: self._generate_three_character,
            4: self._generate_four_character
        }
        
        return generators[self.length]()
    
    def _is_high_quality(self, name: str) -> bool:
        """评估名称质量"""
        quality_score = evaluate_name_quality(name, self.style)
        return quality_score >= 80 and name not in config.FORBIDDEN_NAMES
    
    def _resolve_conflict(self, name: str) -> str:
        """解决名称冲突"""
        # 替换其中一个字
        chars = list(name)
        # 随机选择一个位置替换
        pos = random.randint(0, len(chars) - 1)
        
        # 选择替换字符
        if pos == 0:
            # 替换前缀
            replacement = random.choice(self.word库["prefix"]) if self.word库["prefix"] else random.choice(self.word库["suffix"])
        elif pos == len(chars) - 1:
            # 替换后缀
            replacement = random.choice(self.word库["suffix"])
        else:
            # 替换中间词
            replacement = random.choice(self.word库["middle"]) if self.word库["middle"] else random.choice(self.word库["suffix"])
        
        # 确保替换后的名称长度正确
        if len(replacement) > 1:
            # 如果替换字符是多字，需要调整位置
            chars[pos:] = list(replacement)[:len(chars) - pos]
        else:
            chars[pos] = replacement
        
        return ''.join(chars[:self.length])
    
    def generate_name(self) -> Dict[str, any]:
        """生成单个名称（带质量评估）"""
        max_attempts = 100
        for _ in range(max_attempts):
            name = self._generate_single_name()
            
            # 检查唯一性
            if is_unique(name, self.used_names):
                # 检查质量
                if self._is_high_quality(name):
                    # 添加到已用名称集合
                    add_to_unique(name, self.used_names)
                    # 生成寓意和适用场景
                    meaning = self._generate_meaning(name)
                    usage = self._generate_usage()
                    quality_score = evaluate_name_quality(name, self.style)
                    
                    return {
                        "name": name,
                        "style": self.style,
                        "meaning": meaning,
                        "quality_score": quality_score,
                        "usage_scenarios": usage
                    }
            
            # 尝试解决冲突
            name = self._resolve_conflict(name)
            if is_unique(name, self.used_names) and self._is_high_quality(name):
                add_to_unique(name, self.used_names)
                meaning = self._generate_meaning(name)
                usage = self._generate_usage()
                quality_score = evaluate_name_quality(name, self.style)
                
                return {
                    "name": name,
                    "style": self.style,
                    "meaning": meaning,
                    "quality_score": quality_score,
                    "usage_scenarios": usage
                }
        
        raise RuntimeError("Failed to generate a valid name after multiple attempts")
    
    def _generate_meaning(self, name: str) -> str:
        """生成名称寓意"""
        meanings = {
            "wuxia": [
                "风度翩翩，侠义心肠",
                "武功高强，正气凛然",
                "江湖豪杰，义薄云天",
                "飘逸出尘，侠骨柔情"
            ],
            "qiongyao": [
                "温婉贤淑，柔情似水",
                "浪漫多情，才华横溢",
                "清新脱俗，气质高雅",
                "温柔善良，心灵纯净"
            ],
            "sci-fi": [
                "宇宙浩瀚，未来无限",
                "科技先锋，创新无限",
                "星际探索，勇往直前",
                "智慧超群，远见卓识"
            ],
            "internet": [
                "商业领袖，远见卓识",
                "科技创新，引领潮流",
                "创业先锋，勇于担当",
                "团队协作，共赢未来"
            ]
        }
        
        return random.choice(meanings[self.style])
    
    def _generate_usage(self) -> List[str]:
        """生成适用场景"""
        usages = {
            "wuxia": ["武侠小说", "游戏角色", "影视角色", "笔名"] ,
            "qiongyao": ["言情小说", "影视角色", "笔名", "网名"],
            "sci-fi": ["科幻小说", "游戏角色", "科技产品", "笔名"],
            "internet": ["公司名称", "产品名称", "花名", "品牌名称"]
        }
        
        return random.sample(usages[self.style], 2)
    
    def batch_generate(self) -> List[Dict[str, any]]:
        """批量生成名称"""
        names = []
        for _ in range(self.count):
            names.append(self.generate_name())
        return names
```

### 3.3 唯一性检查模块 (uniqueness.py)

```python
from typing import Set
from pybloom_live import BloomFilter
from app.config import config

# 全局布隆过滤器
bloom_filter = BloomFilter(capacity=config.BLOOM_FILTER_SIZE, error_rate=config.BLOOM_FILTER_ERROR_RATE)

# 全局已用名称集合
global_used_names = set()

def is_unique(name: str, local_used_names: Set[str]) -> bool:
    """检查名称是否唯一"""
    # 1. 检查本地已用名称
    if name in local_used_names:
        return False
    
    # 2. 检查全局已用名称
    if name in global_used_names:
        return False
    
    # 3. 检查布隆过滤器
    if name in bloom_filter:
        # 布隆过滤器可能有误判，需要进一步确认
        # 这里简化处理，直接返回False
        return False
    
    return True

def add_to_unique(name: str, local_used_names: Set[str]):
    """添加名称到唯一集合"""
    local_used_names.add(name)
    global_used_names.add(name)
    bloom_filter.add(name)

def get_conflict_rate(generated_count: int) -> float:
    """计算冲突率"""
    if generated_count == 0:
        return 0.0
    
    # 理论冲突率计算
    # 布隆过滤器误判率 + 全局集合重复率
    return config.BLOOM_FILTER_ERROR_RATE
```

### 3.4 质量评估模块 (quality.py)

```python
from pypinyin import pinyin, Style
from typing import List

# 常用姓氏列表（用于互联网风格评估）
COMMON_SURNAMES = ["王", "李", "张", "刘", "陈", "杨", "赵", "黄", "周", "吴"]

# 常用词列表（用于评估常用度）
COMMON_WORDS = ["明", "强", "伟", "丽", "芳", "军", "涛", "勇", "艳", "杰"]

def _get_pinyin(name: str) -> List[str]:
    """获取名称拼音"""
    return [" ".join([p[0] for p in pinyin(char, style=Style.NORMAL)]) for char in name]

def _evaluate_phonetic(name: str) -> float:
    """评估音韵（40分）"""
    score = 40
    
    # 1. 检查是否有不雅谐音
   不雅_homophones = ["死", "屎", "尿", "傻", "笨", "蠢"]
    pinyin_list = _get_pinyin(name)
    flat_pinyin = "".join(pinyin_list)
    
    for bad in 不雅_homophones:
        if bad in flat_pinyin:
            score -= 20
            break
    
    # 2. 检查读音流畅度（避免相同或相似的拼音连续出现）
    for i in range(len(pinyin_list) - 1):
        if pinyin_list[i] == pinyin_list[i+1]:
            score -= 10
    
    # 3. 检查平仄搭配（简化处理：避免全平或全仄）
    # 这里简化处理，只检查是否有变化
    if len(set(pinyin_list)) < 2:
        score -= 10
    
    return max(0, score)

def _evaluate_meaning(name: str, style: str) -> float:
    """评估寓意（30分）"""
    score = 30
    
    # 根据风格检查关键词
    style_keywords = {
        "wuxia": ["风", "云", "剑", "侠", "龙", "虎", "天", "地"],
        "qiongyao": ["雨", "雪", "梦", "花", "月", "星", "心", "情"],
        "sci-fi": ["星", "宇", "航", "云", "风", "电", "光", "天"],
        "internet": ["云", "风", "雨", "电", "光", "天", "地", "人"]
    }
    
    # 检查名称中是否包含风格关键词
    has_keyword = any(keyword in name for keyword in style_keywords[style])
    if not has_keyword:
        score -= 15
    
    # 检查名称长度是否符合风格特点
    if style == "internet" and len(name) == 2:
        # 互联网风格双字名更符合创始人风格
        score += 5
    
    return max(0, score)

def _evaluate_commonness(name: str) -> float:
    """评估常用度（30分）"""
    score = 30
    
    # 1. 检查是否包含常用词
    common_word_count = sum(1 for char in name if char in COMMON_WORDS)
    score -= common_word_count * 5
    
    # 2. 检查是否为常见姓名结构
    if len(name) == 2 and name[0] in COMMON_SURNAMES and name[1] in COMMON_WORDS:
        # 非常常见的姓名结构
        score -= 10
    
    return max(0, score)

def evaluate_name_quality(name: str, style: str) -> float:
    """评估名称质量"""
    from app.config import config
    
    # 1. 音韵评估
    phonetic_score = _evaluate_phonetic(name)
    
    # 2. 寓意评估
    meaning_score = _evaluate_meaning(name, style)
    
    # 3. 常用度评估
    commonness_score = _evaluate_commonness(name)
    
    # 综合评分
    total_score = (phonetic_score * config.PHONETIC_WEIGHT +
                  meaning_score * config.MEANING_WEIGHT +
                  commonness_score * config.COMMONNESS_WEIGHT)
    
    return round(total_score, 2)
```

### 3.5 FastAPI应用入口 (main.py)

```python
from fastapi import FastAPI, HTTPException, Query
from app.generator import NameGenerator
from typing import List

app = FastAPI(title="中文名称自动生成API", version="1.0.0")

@app.get("/generate", summary="生成单个名称")
def generate_name(
    style: str = Query(..., description="名称风格：wuxia/qiongyao/sci-fi/internet"),
    length: int = Query(2, ge=2, le=4, description="名称长度：2-4个汉字")
):
    """生成单个中文名称"""
    try:
        generator = NameGenerator(style=style, length=length, count=1)
        name = generator.generate_name()
        return name
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))
    except RuntimeError as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/batch_generate", summary="批量生成名称")
def batch_generate(
    style: str = Query(..., description="名称风格：wuxia/qiongyao/sci-fi/internet"),
    length: int = Query(2, ge=2, le=4, description="名称长度：2-4个汉字"),
    count: int = Query(10, ge=1, le=100, description="生成数量：1-100")
):
    """批量生成中文名称"""
    try:
        generator = NameGenerator(style=style, length=length, count=count)
        names = generator.batch_generate()
        return {"names": names}
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))
    except RuntimeError as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/health", summary="健康检查")
def health_check():
    """检查API健康状态"""
    return {"status": "healthy", "message": "中文名称自动生成API运行正常"}
```

import pytest
from app.generator import NameGenerator
